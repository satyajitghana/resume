% --- Build toggles -------------------------------------------------------------
%  Wrappers define these (before \input-ing this file) to switch modes:
%    \ATSMODE   -> suppress the kestrel watermark (clean ATS build)
%    \ONEPAGER  -> compact sizing + drop \verbose{} content to fit one page
\newif\ifonepager
\ifdefined\ONEPAGER\onepagertrue\fi

% --- Packages ----------------------------------------------------------------
\usepackage[a4paper,top=1.35cm,bottom=1.2cm,left=1.5cm,right=1.5cm]{geometry}
\usepackage{fontspec}
\usepackage{xcolor}
\usepackage{enumitem}
\usepackage{titlesec}
\usepackage{graphicx}
\usepackage{eso-pic}
\usepackage{microtype}
\usepackage[hidelinks]{hyperref}

% --- Palette -----------------------------------------------------------------
\definecolor{ink}{HTML}{1A1F26}      % body text — near-black, slate-tinted
\definecolor{head}{HTML}{0F141A}     % headings — darkest
\definecolor{accent}{HTML}{B45309}   % rust/ochre — echoes the kestrel
\definecolor{muted}{HTML}{6B7280}    % dates, secondary detail
\definecolor{rule}{HTML}{D6D3CE}     % hairline rules

% --- Fonts -------------------------------------------------------------------
\defaultfontfeatures{Ligatures=TeX}

% NOTE: RawFeature={-calt} disables Inter's contextual alternates, which otherwise
% remap glyphs like "+" to Private-Use codepoints that break text extraction (ATS).
\setmainfont{Inter}[
  Path = fonts/, Extension = .ttf, RawFeature = {-calt},
  UprightFont = *-Regular, ItalicFont = *-Italic,
  BoldFont = *-SemiBold, BoldItalicFont = *-SemiBoldItalic ]

\newfontfamily\display{SpaceGrotesk}[
  Path = fonts/, Extension = .ttf, RawFeature = {-calt},
  UprightFont = *-Medium, BoldFont = *-Bold ]
\newcommand*{\displaylight}{\fontspec{SpaceGrotesk}[Path=fonts/,Extension=.ttf,RawFeature={-calt},UprightFont=*-Light]}

\newcommand*{\bodylight}{\fontspec{Inter}[Path=fonts/,Extension=.ttf,RawFeature={-calt},UprightFont=*-Light]}
\newcommand*{\bodymedium}{\fontspec{Inter}[Path=fonts/,Extension=.ttf,RawFeature={-calt},UprightFont=*-Medium]}

\color{ink}
\setlength{\parindent}{0pt}
\frenchspacing

% --- Sizing parameters (defaults = 2-page build) -----------------------------
\def\NAMESIZE{30}   \def\TAGSIZE{11}   \def\CONTACTSIZE{8.6}
\def\SECSIZE{11.5}  \def\ENTRYSIZE{10.8}
\def\SECABOVE{10pt} \def\SECBELOW{5pt}
\def\ENTRYSKIP{3pt} \def\AFTERENTRY{2.5pt}
\def\ITEMSEP{1.6pt} \def\TOPSEP{2pt}
\def\ROWSKIP{2.6pt} \def\LEAD{1.04}

% --- One-page build: compact sizing to fit the full content densely on one page,
%     dropping only \verbose{} extras (currently just the coursework line) --------
\ifonepager
  \geometry{top=0.7cm,bottom=0.6cm,left=1.1cm,right=1.1cm}
  \def\NAMESIZE{20}  \def\TAGSIZE{8.0} \def\CONTACTSIZE{7.0}
  \def\SECSIZE{8.4}  \def\ENTRYSIZE{8.4}
  \def\SECABOVE{3.4pt}\def\SECBELOW{1.6pt}
  \def\ENTRYSKIP{1.5pt}\def\AFTERENTRY{0.9pt}
  \def\ITEMSEP{0.7pt}\def\TOPSEP{0.9pt}
  \def\ROWSKIP{1.4pt}\def\LEAD{1.0}
  \renewcommand{\footnotesize}{\fontsize{6.4}{7.9}\selectfont}% stack lines
  \renewcommand{\small}{\fontsize{6.7}{8.3}\selectfont}%       skills/dates
  \renewcommand{\normalsize}{\fontsize{7.0}{8.6}\selectfont}\normalsize
\fi
\linespread{\LEAD}

% Content toggles between the readable long resume and the crammed one-pager:
%   \verbose{...}     -> appears only on the LONG resume (dropped on the one-pager)
%   \onepageonly{...} -> appears only on the ONE-PAGER (keeps the long resume to 2 pages)
\newcommand{\verbose}[1]{\ifonepager\else#1\fi}
\newcommand{\onepageonly}[1]{\ifonepager#1\fi}

% --- Hyperlinks (kept as real, selectable text) ------------------------------
\hypersetup{

  pdftitle = {Satyajit Ghana — Resume}, pdfauthor = {Satyajit Ghana} }

% ============================================================================
%  Section headings
% ============================================================================
\titlespacing*{\section}{0pt}{\SECABOVE}{\SECBELOW}
\titleformat{\section}
  {\display\fontsize{\SECSIZE}{13}\selectfont\color{head}\addfontfeatures{LetterSpace=8.0}}
  {}{0pt}
  {\MakeUppercase}
  [{\color{rule}\titlerule[0.8pt]}]

% ============================================================================
%  Building blocks
% ============================================================================

% \entry{title}{right-aligned meta}{subtitle / role}{location or stack}
\newcommand{\entry}[4]{%
  \vspace{\ENTRYSKIP}%
  \noindent\begin{minipage}[t]{\textwidth}%
    {\bodymedium\fontsize{\ENTRYSIZE}{13}\selectfont\color{head}#1}\hfill
    {\small\color{muted}#2}\\[1.5pt]
    {\itshape\color{accent}#3}\hfill{\small\itshape\color{muted}#4}%
  \end{minipage}\par\vspace{\AFTERENTRY}}

% Tight, accent-marked bullet list for achievements
\newlist{points}{itemize}{1}
\setlist[points]{
  leftmargin = 1.2em, itemsep = \ITEMSEP, topsep = \TOPSEP, parsep = 0pt,
  label = {\color{accent}\footnotesize\textbullet}}   % real glyph: extracts for ATS

% Tech-stack line at the foot of an entry
\newcommand{\tools}[1]{\vspace{1.5pt}\par{\footnotesize\color{muted}%
  {\bodymedium\color{head}Stack\,\,}\textbar\,\,#1}\par}

% Skills row:  \skillrow{Category}{items}
\newcommand{\skillrow}[2]{%
  \noindent{\small{\bodymedium\color{head}#1}\,\,{\color{muted}\textbar}\,\, #2}\par\vspace{\ROWSKIP}}

% Inline separator for the contact line
\newcommand{\sep}{\hspace{3pt}{\color{rule}\textbullet}\hspace{3pt}}

% ============================================================================
%  Header
% ============================================================================
\newcommand{\makeheader}[8]{% name, tagline, location, email, phone, site, github, linkedin
  {\display\fontsize{\NAMESIZE}{\NAMESIZE}\selectfont\color{head}\addfontfeatures{LetterSpace=1.0}#1}\par
  \vspace{3pt}
  {\bodylight\fontsize{\TAGSIZE}{14}\selectfont\color{accent}\addfontfeatures{LetterSpace=2.0}#2}%
  \hfill{\small\color{muted}#3}\par
  \vspace{6pt}
  {\fontsize{\CONTACTSIZE}{11}\selectfont\color{ink}%
    \href{mailto:#4}{#4}\sep
    #5\sep
    \href{https://#6}{#6}\sep
    \href{https://#7}{#7}\sep
    \href{https://#8}{#8}}\par
  \vspace{5pt}
  {\color{accent}\rule{\linewidth}{1.2pt}}\par
  \vspace{2pt}}
